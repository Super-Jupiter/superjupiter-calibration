cmake_minimum_required(VERSION 3.16)
project(SuperJupiterCalibrationConsumer LANGUAGES CXX)
if(DEFINED SUPERJUPITER_CALIBRATION_SOURCE_DIR)
  add_subdirectory("${SUPERJUPITER_CALIBRATION_SOURCE_DIR}" calibration)
  if(TARGET superjupiter-calibration-tests OR TARGET superjupiter-calibration-replay)
    message(FATAL_ERROR "A consumer unexpectedly builds library developer tools")
  endif()
else()
  find_package(SuperJupiterCalibration 0.1 CONFIG REQUIRED)
endif()
add_executable(superjupiter-calibration-consumer main.cpp)
target_link_libraries(superjupiter-calibration-consumer PRIVATE SuperJupiter::Calibration)
